# rlpsij (shared/comps/rlpsij.c) on the simulated drive, booted from the
# bench config, wired to hv0 like conf/template/id_pmsm.txt wires idpmsm,
# plus hv0.vel = rlpsij0.com_vel for the current loop's predictor.
# rlpsij has no fb_rev input, so it reads the true shaft angle here.
# Light motor, differs from the config the current loop runs with:
# r 1.8, psi 0.06, j 0.00003, f 0.02, d 0.0001 (l 0.022957 from conf).
# Expected: state 6, r ~ 1.8, l ~ 0.023, polecount 5, psi ~ 0.06,
# j ~ 0.00003, friction ~ 0.02; speed stays within ~ +-65 rad/s
# (max_vel 62.8) and ends at 0. host/tests/motor_*.test check this on
# 20 motors.
hv0.sim_r = 1.8
hv0.sim_psi = 0.06
hv0.sim_j = 0.00003
hv0.sim_f = 0.02
hv0.sim_d = 0.0001
load rlpsij
rlpsij0.rt_prio = 4
pid0.en = 0
hv0.en = rlpsij0.en_out
hv0.cmd_mode = rlpsij0.cmd_mode
hv0.d_cmd = rlpsij0.d_cmd
hv0.q_cmd = rlpsij0.q_cmd
hv0.pos = rlpsij0.com_pos
rlpsij0.ud_fb = hv0.ud_fb
rlpsij0.uq_fb = hv0.uq_fb
rlpsij0.id_fb = hv0.id_fb
rlpsij0.iq_fb = hv0.iq_fb
rlpsij0.abs_pos_fb = hv0.sim_pos
hv0.vel = rlpsij0.com_vel
rlpsij0.test_cur = 2
relink
start
debug_level 0
fault0.cmd_error = 0
rlpsij0.en = 1
run 16

rlpsij0.r
rlpsij0.l
