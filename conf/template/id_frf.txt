# measure the coupling with a stepped-sine
# FRF (comp frf): fits a two-mass model and
# prints conf0.j_lpf (and the resonance for a
# biquad notch, link filters). the axis must
# be free to move: it creeps back and forth
# at frf0.creep_vel between frf0.min_pos and
# frf0.max_pos about where it starts (rad),
# measures only while cruising, ends where it
# started. the lowest point needs one cruise:
# (max_pos - min_pos) / creep_vel above ~3 s
# at f_start 5 Hz. starts on enable (start
# follows the pid enable); disable to abort or
# clear. run it before id_pid: j_lpf changes
# the plant the gains are tuned against.
load frf
frf0.rt_prio = 8.6
frf0.en = fault0.en_out
frf0.start = fault0.en_pid
frf0.pos_fb = fb_switch0.pos_fb
pid0.pos_ext_cmd = frf0.pos_cmd
pid0.vel_ext_cmd = frf0.vel_cmd
pid0.acc_ext_cmd = frf0.acc_cmd

frf0.in = pid0.torque_cmd
pmsm_ttc0.torque = frf0.out
frf0.pos = fb_switch0.vel_fb
vel1.torque = frf0.out
vel2.torque = frf0.out

frf0.amp = 1
frf0.creep_vel = 3
frf0.creep_acc = 20
frf0.min_pos = -10
frf0.max_pos = 10

conf0.max_pos_error = 0

term0.wave0 = frf0.freq
term0.gain0 = 1
term0.wave1 = frf0.inj
term0.gain1 = 20
